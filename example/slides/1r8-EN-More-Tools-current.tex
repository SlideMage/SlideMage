% SlideMage manual, appendix: the tools not covered in the chapters
% (installation has its own chapter, "Installing SlideMage").
\setSlideVersion{1-1-1}
\begin{standardframe}{More tools}
	\small
	Tools you need now and then, all in SlideMage's \textcolor{c_primary_highlight}{three-dot menu} and the \textcolor{c_primary_highlight}{command palette} (\SMCode{Ctrl+K}).
	\par\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.48\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item \textbf{Import PPTX/ODP} (\emph{Decks}) turns the slides of an existing presentation into a SlideMage deck, for you to finish by hand
				\item \textbf{Housekeeping}: archive unused content, update slide previews, reestablish translation links, empty the \SMCode{pdf} folder, remove temporary files
				\item \textbf{Find anything} (\SMCode{Ctrl+K}) finds a command, a slide by its ID or name, or a word in the slides, decks and bibliography
			\end{itemize}
		\end{column}
		\begin{column}{0.48\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item \textbf{Settings} (\emph{Project}): dark mode, the default template for single-slide renders, AI models, cloud use, prompts, updates
				\item \textbf{Projects}: \emph{Open project\ldots} switches project; \SMCode{.slidemage/project.json} records which project SlideMage last opened; keep it with the project
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
