% SlideMage manual, chapter "Working in SlideMage": the deck editor, showing
% this manual's own deck (tools/manual/make_screenshots.py).
\setSlideVersion{1-0-0}
\begin{standardframe}{The deck editor}
	\small
	Click a deck's name in the main window to open the deck editor: one row per slide and per chapter, in deck order. Everything you do here -- adding, moving,	removing slides and chapters -- is written into the deck file at once (with an \emph{Undo} option), so you never need to edit it by hand.
	\par\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.56\linewidth}
			{\setlength{\fboxsep}{0pt}\fcolorbox{c_primary_gray}{white}{\includegraphics[width=\dimexpr\linewidth-0.8pt\relax]{media/screenshot-slidemage-deck-editor-list.png}}}
		\end{column}
		\begin{column}{0.41\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item \emph{+ before} / \emph{+ after} open the \emph{Add slide} dialog: a new slide, an existing one, or a copy of one
				\item The dropdown chooses how the slide is included (\SMCmd{slide}, \SMCmd{slidestep}, \ldots)
				\item Move a slide by typing its new number; \emph{Undo} takes back the last change
				\item Views: \emph{List}, \emph{Thumbnail}, \emph{Grid}
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
