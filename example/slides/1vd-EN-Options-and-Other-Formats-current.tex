\begin{standardframe}{Options and other formats}
	\small
	\emph{Build options\ldots} (\emph{Build} menu, or the arrow of \emph{Compile PDF}) changes how PDFs are made; other outputs have entries of their own.
	\par
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.63\linewidth}
			\textbf{Build options}
			\shrinklists{3}
			\begin{itemize}
				\item \emph{Generate appendix TOC}: an agenda and clickable slide previews after every whole-deck PDF (also a deck setting)
				\item \emph{Pages replace theirs in the full PDF}: see \emph{Compiling single pages}
				\item \emph{Auto-open generated PDFs}: opens each finished PDF
			\end{itemize}
			\textbf{Other outputs}
			\shrinklists{3}
			\begin{itemize}
				\item \emph{Export PPTX\ldots} (\emph{Deck} $\rightarrow$ \emph{Export}): a PowerPoint file for recording a lecture, reveal steps fading in one by one
					\item \emph{Export LaTeX\ldots} (\emph{Deck} $\rightarrow$ \emph{Export}): a folder that compiles anywhere with \SMCode{latexmk}
				\item \emph{Write transcript} (\emph{Build} $\rightarrow$ \emph{Accessible transcript}): the deck as text
			\end{itemize}
		\end{column}
		\begin{column}{0.34\linewidth}
			\centering
			{\setlength{\fboxsep}{0pt}\fcolorbox{c_primary_gray}{white}{\includegraphics[width=\dimexpr\linewidth-0.8pt\relax]{media/screenshot-slide-appendix-grid.png}}}\par
			{\scriptsize the appendix: a grid of slide previews\par}
		\end{column}
	\end{columns}
\end{standardframe}
