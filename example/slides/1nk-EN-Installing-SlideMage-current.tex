% SlideMage manual, chapter "Installing SlideMage": the app itself, its first
% start and its updates.
\setSlideVersion{1-0-0}
\begin{standardframe}{Installing SlideMage}
	\small
	SlideMage is a single Windows program without Python. The GitHub
	repository is private: your account needs access for download and updates.
	\par\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.49\linewidth}
			\shrinklists{1}
			\textbf{Download and first start}
			\begin{itemize}
				\item Get \SMCode{SlideMage-windows.zip} from the newest \emph{Release}
				\item Unzip and start \SMCode{SlideMage.exe} (Windows warning: \emph{More info} $\rightarrow$ \emph{Run anyway})
				\item Choose a project folder; an empty one becomes a new project
			\end{itemize}
		\end{column}
		\begin{column}{0.47\linewidth}
			\shrinklists{1}
			\textbf{Updates and more}
			\begin{itemize}
				\item \emph{Settings} $\rightarrow$ General $\rightarrow$ \emph{Updates}: GitHub token (read access), check/ask/install
				\item Framework and manual are copies; a newer SlideMage offers to update them
				\item From source: Python 3.11+, \SMCode{git clone}, \SMCode{pip install -e .}
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
