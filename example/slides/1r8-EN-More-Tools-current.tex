% SlideMage manual, appendix: the tools not covered in the chapters
% (installation has its own chapter, "Installing SlideMage").
\setSlideVersion{1-1-1}
\begin{standardframe}{More tools}
	\small
	Tools you need now and then, all in SlideMage's \textcolor{c_primary_highlight}{menus}.
	\par\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.48\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item \textbf{Import PPTX/ODP} (Decks menu) turns the slides of an existing presentation into a SlideMage deck, for you to finish by hand
				\item \textbf{Housekeeping}: archive unused content, update slide previews, sync translation links, empty the \SMCode{pdf} folder, remove temporary files
				\item \textbf{Slide name search} and \textbf{Project-wide full-text search} in the main window find any slide by its ID or name, or by its text
			\end{itemize}
		\end{column}
		\begin{column}{0.48\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item \textbf{Preferences} (System menu): dark mode, the default template for single-slide renders, AI models and prompts, updates
				\item \textbf{Projects}: \SMCode{.slidemage/project.json} records which project SlideMage last opened; keep it with the project
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
