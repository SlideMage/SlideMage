% SlideMage manual, chapter "Content and layout": placing figures. The slide
% itself is a besidefigure: the figure in the top-right corner with its
% border on the margins, the text column exactly as wide as the room left.
\setSlideVersion{1-0-0}
\begin{standardframe}{Placing figures}
	\begin{besidefigure}[height=4.2cm, border]{media/illustration-bar-chart.pdf}
		{\small\SMLineTop A figure needs no frame of its own: these commands place
		it on any slide, aligned with the margins.\par}
		\smallskip
		\shrinklists{1}
		\begin{itemize}
			\item \SMCmd{cornerfigure}: in a corner (\SMCode{ne}, \SMCode{se}, \ldots), sized by width \emph{or} height
			\item \SMCode{besidefigure}: text beside it (this slide)
			\item \SMCmd{fitfigure}: as large as fits below the text
			\item \SMCode{border}: the border's outer edge on the margins
		\end{itemize}
		\SMCodeFile{snippets/1c9-1.tex}
	\end{besidefigure}
\end{standardframe}
