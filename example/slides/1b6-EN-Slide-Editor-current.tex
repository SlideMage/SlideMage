\begin{standardframe}{The slide editor}
	\small
	A slide's source opens in the slide editor, from the deck editor or from
	\emph{Find slide}. You edit the LaTeX directly, with syntax colors and a
	toolbar for the commands you need most, and see the result without
	compiling the whole deck.
	\par\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.56\linewidth}
			{\setlength{\fboxsep}{0pt}\fcolorbox{c_primary_gray}{white}{\includegraphics[width=\dimexpr\linewidth-0.8pt\relax]{media/screenshot-slidemage-slide-editor.png}}}
		\end{column}
		\begin{column}{0.41\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item \emph{Render to PDF} compiles the editor's text, saved or not
				\item \emph{Save} asks for the version (see \emph{Slide versions})
				\item Duplicate, rename, delete; language; transcript summary
				\item With a local AI model: \emph{Check syntax}, \emph{AI draft}, shown as a diff you accept or reject
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
