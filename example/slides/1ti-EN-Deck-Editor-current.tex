% SlideMage manual, chapter "Working in SlideMage": the slide list (the workspace's left
% column), shown in a short window with three slides selected
% (tools/manual/make_screenshots.py, shot slide-list).
\setSlideVersion{1-0-1+}
\providecommand{\callout}[2]{\node[circle, fill=c_primary_highlight, text=white, inner sep=1pt, minimum size=9pt, font=\tiny\bfseries] at (#1) {#2};}
\begin{standardframe}{The slide list}
	\small
	The left column lists the open deck's slides in sections. Every change here is written into the deck file at once, and \SMCode{Ctrl+Z} (or \emph{Undo} on the message that appears) takes it back.
	\par
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.21\linewidth}
			\begin{tikzpicture}
				\node[anchor=south west, inner sep=0pt] (shot) at (0,0) {\setlength{\fboxsep}{0pt}\fcolorbox{c_primary_gray}{white}{\includegraphics[width=\dimexpr\linewidth-0.8pt\relax]{media/screenshot-slidemage-slide-list.png}}};
				\begin{scope}[x={(shot.south east)}, y={(shot.north west)}]
					% beside each element, as fractions of the picture (the list in a short window)
					\callout{0.946,0.955}{1}
					\callout{0.800,0.884}{2}
					\callout{0.911,0.807}{3}
					\callout{0.508,0.128}{4}
					\callout{0.578,0.313}{5}
				\end{scope}
			\end{tikzpicture}
		\end{column}
		\begin{column}{0.76\linewidth}
			\shrinklists{3}
			\begin{enumerate}
				\item \emph{Find slide}: part of an ID or title; \SMCode{Esc} clears it
				\item \emph{Not reviewed}: slides the assistant changed that you have not checked yet
				\item A row: thumbnail, number, ID, language, title. A red dot is a source error; notes say \emph{hidden}, \emph{no DE}, \emph{preview old}; a chip shows how the deck calls it
				\item \SMCode{Ctrl}- or \SMCode{Shift}-click selects several; the bar shows or hides them in this view, or removes them
				\item A section heading folds on a click; \SMCode{F2} renames it; right-click $\rightarrow$ \emph{Remove section\ldots}: only it, it with its slides (their files stay), or every section
			\end{enumerate}
			\begin{itemize}
				\item A click shows the slide; a double click or \SMCode{Enter} opens its source
				\item Drag to move, also between sections; \SMCode{Alt+Up}/\SMCode{Alt+Down} move a step; \SMCode{Delete} takes it out
				\item Right-click: \emph{Call as} (live, final, pinned, \ldots), visibility, add, duplicate, move, \emph{Update versions\ldots}
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
