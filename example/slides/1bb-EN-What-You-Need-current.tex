% SlideMage manual, chapter "Installing SlideMage": the overview of what to
% install. The next slides cover each program.
\setSlideVersion{1-2-0}
\begin{standardframe}{What you need}
	\small
	SlideMage is the app you work in; it calls LaTeX to typeset the PDFs. The first three programs below are required, the rest are optional. The following slides cover each one.
	\par\medskip
	{\footnotesize
	\begin{SMTable}[gap=12pt]{lll}
		\toprule
		Program & What it is for & Where to get it \\
		\midrule
		MiKTeX or TeX Live & typesetting: LuaLaTeX, biber, latexmk & \SMCode{miktex.org}, \SMCode{tug.org/texlive} \\
		Perl (with MiKTeX only) & runs latexmk & Git for Windows, \SMCode{git-scm.com} \\
		SlideMage & the app & its releases on GitHub \\
		\midrule
		\emph{Optional:} Ollama and models & descriptions, translation, chat & \SMCode{ollama.com} \\
		\emph{Optional:} Claude Code & second assistant tab & see \emph{Optional helpers} \\
		\emph{Optional:} viewer, pympress, \ldots & reading, presenting & see \emph{Optional helpers} \\
		\emph{Optional:} Python 3.11+ & running from source & \SMCode{python.org} \\
		\bottomrule
	\end{SMTable}\par}
\end{standardframe}
