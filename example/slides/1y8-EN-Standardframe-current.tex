\begin{standardframe}{The default frame: \SMCode{standardframe}}
	{\small Almost every slide is a \SMCode{standardframe}: you give it a title
	and fill it with text, lists, figures or code, and the framework adds the
	title bar, logo and footer and places the content. Left, its output; right,
	its source.\par}
	\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.46\linewidth}
			\begin{itemize}
				\item Title bar, logo, footer and bullets come with the frame
				\begin{itemize}
					\item Sub-items work as usual
				\end{itemize}
				\item Content is top-aligned; the first line sits on the Top Margin
			\end{itemize}
		\end{column}
		\begin{column}{0.5\linewidth}
			\SMCodeFile{snippets/1y8-1.tex}
		\end{column}
