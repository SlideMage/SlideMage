\begin{standardframe}{Slide versions}
	\begin{itemize}
		\item \SMCode{...-current.tex} is the live slide; \SMCode{...-v1-2-0.tex} files are frozen snapshots, never edited again
		\item Every \emph{Save} in SlideMage's slide editor asks:
	\end{itemize}
	\smallskip
	\hspace*{\SMItemStep}{\setlength{\fboxsep}{0pt}\fcolorbox{c_primary_gray}{white}{\includegraphics[width=0.55\linewidth]{media/screenshot-slidemage-save-dialog.png}}}
	\smallskip
	\begin{itemize}
		\item Major, Minor, Patch: the old content is frozen first, then the new one gets the higher number; \emph{Overwrite}: just save (a typo)
		\item A reveal-only change is a patch; the first version is 1.0.0
		\item A deck can pin a version: \SMCmd{slideversion}\SMCode{\{1an-EN-Slide-Versions\}\{1-0-0\}}
	\end{itemize}
\end{standardframe}
