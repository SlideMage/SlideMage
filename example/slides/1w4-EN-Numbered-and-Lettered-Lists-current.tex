\begin{standardframe}{Numbered and lettered lists}
	{\small Besides bullets, the framework sets numbered and lettered lists in the same style:
	\SMCode{1)} or \SMCode{1.} for numbers, \SMCode{a)} or \SMCode{a.} for letters; a plain
	\SMCode{enumerate} works too. On a crowded slide, \SMCmd{shrinklists}\SMCode{\{n\}} sets every
	list level \emph{n} font sizes smaller; 3 is the smallest.\par}
	\vspace{-2pt}
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.44\linewidth}
			\begin{SMNumItemize}
				\item Numbered: 1) 2) \ldots
				\item Second
			\end{SMNumItemize}
			\begin{SMNumPeriodItemize}
				\item Numbered: 1. 2. \ldots
				\item Second
			\end{SMNumPeriodItemize}
			\begin{SMAlphaItemize}
				\item Lettered: a) b) \ldots
				\item Second
			\end{SMAlphaItemize}
			\begin{SMAlphaPeriodItemize}
				\item Lettered: a. b. \ldots
				\item Second
			\end{SMAlphaPeriodItemize}
		\end{column}
		\begin{column}{0.53\linewidth}
			\SMCodeFile{snippets/1w4-1.tex}
		\end{column}
	\end{columns}
\end{standardframe}
