\begin{standardframe}{Anatomy of a deck file}
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.55\linewidth}
			\SMCodeFile[basicstyle=\SMCodeFont\fontsize{6.4}{7.6}\selectfont]{snippets/1y5-1.tex}
		\end{column}
		\begin{column}{0.42\linewidth}
			\small
			A deck is a LaTeX file in \SMCode{decks/}: a template, a few settings,
			then the slides in chapters. SlideMage writes most of it for you.
			\par\medskip
			\shrinklists{2}
			\begin{itemize}
				\item The first lines choose LuaLaTeX and the template
				\item The version block lets SlideMage compile each deck version
				\item \SMCmd{decklanguage}, \SMCmd{decktags} and \SMCmd{deckoutputfolder} are settings SlideMage reads; the course link is the CampusOnline plugin's
				\item Then the slides, in chapters (\SMCmd{section})
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
