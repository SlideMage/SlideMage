% SlideMage manual, chapter "The big picture": the folders of a project.
\setSlideVersion{1-0-0}
\begin{standardframe}{Anatomy of a project}
	\small
	A project is an ordinary folder of plain files, so it can live in a synced
	folder or under Git. The file \SMCode{slide\_ids.csv} at its top marks it as a
	project; open it with System $\rightarrow$ \emph{Open Project}. SlideMage keeps
	the folders in order; you rarely need to open them yourself.
	\par\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.5\linewidth}
			\SMCodeFile{snippets/1jv-1.tex}
		\end{column}
		\begin{column}{0.46\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item On start, SlideMage opens the project given by \SMCode{--project}, else the one above the current folder, else the last one used
				\item The LaTeX framework is not part of the project: it sits beside it, in \SMCode{latex/}
			\end{itemize}
			\medskip
			\begin{infoblock}{Generated, never edited}
				\SMCode{pdf/}, \SMCode{.build/}, \SMCode{.previews/}, \SMCode{transcripts/}, \SMCode{media/alt-text.csv}
			\end{infoblock}
		\end{column}
	\end{columns}
\end{standardframe}
