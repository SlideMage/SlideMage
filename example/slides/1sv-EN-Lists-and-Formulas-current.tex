\begin{standardframe}{Lists and formulas}
	{\small The framework sets bullets, indents and spacing; you only write
	\SMCmd{item}. An item that holds a tall formula, such as a fraction, uses
	\SMCmd{tallitem}, so the item after it keeps its distance.\par}
	\medskip
	\begin{columns}[T,onlytextwidth]

			\begin{itemize}

				\end{itemize}
				\tallitem Sharpe ratio $S=\dfrac{E[R]-r_f}{\sigma}$
				\tallitem The item after a fraction, too
			\end{itemize}
			\begin{gothamalphaitemize}
				\item Lettered: a), b), \ldots
			\end{gothamalphaitemize}
		\end{column}
		\begin{column}{0.53\linewidth}
			\SMCodeFile{snippets/1sv-1.tex}

		\end{column}
	\end{columns}
\end{standardframe}
