% SlideMage manual, chapter "Working in SlideMage": the checks before and
% after a compile (README: "Pre-render syntax check").
\begin{standardframe}{Safety nets}
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.47\linewidth}
			\scriptsize
			\begin{gothamtable}[gap=8pt]{ll}
				\toprule
				Before every compile: & finds \\
				\midrule
				\SMCode{braces}, \SMCode{environments} & left open \\
				\SMCode{math-dollar}, \SMCode{math-delims} & maths left open \\
				\SMCode{ampersand}, \SMCode{hash} & a stray \SMCode{\&} or \SMCode{\#} \\
				\SMCode{citation} & unknown key \\
				\SMCode{image}, \SMCode{code-file} & a missing file \\
				\SMCode{slide-file} & missing slide \\
				\SMCode{underscore}, \ldots & warnings \\
				\bottomrule
			\end{gothamtable}
		\end{column}
		\begin{column}{0.5\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item An error stops the compile and asks: \emph{Cancel}, \emph{Show in editor} or \emph{Render anyway}; warnings go to the log
				\item After a compile, \SMCode{check\_layout.py} measures the PDF against the margins; a slide can exempt itself with \SMCode{\%\% layout-check: skip top}
				\item \emph{Search all slide texts} finds any word in every slide
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
