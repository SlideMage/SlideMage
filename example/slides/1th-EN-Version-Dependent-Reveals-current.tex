% SlideMage manual, chapter "Reveal steps": reveals that depend on the deck
% version. The list on the left is live: three steps in the instructor deck,
% one page in the student decks (compile the manual as during-course or
% after-course to see it). This slide is also the one shown in three
% versions on slide 1rv ("One deck, several PDFs"). [T]: the table's top,
% not its middle, lines up with the list.
\setSlideVersion{1-0-0}
\begin{standardframe}{Reveals that depend on the deck version}
	{\small A reveal often hides an answer. With these commands a slide unfolds
	step by step in your instructor deck, while students get it complete -- or,
	for a solution, only after the course. The list below is live: compare the
	deck versions.\par}
	\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.34\linewidth}
			\begin{itemize}
				\item A question for the class
				\conditionalpause
				\item The answer: a step for the instructor, there at once for students
				\deckitem<3-> \SMCmd{deckitem}\SMCode{<3->}: the same for one item
				\deckreveal[4]{\item \SMCmd{deckreveal}\SMCode{[4]}: the solution -- left out of the during-course deck}
			\end{itemize}
		\end{column}
		\begin{column}{0.63\linewidth}
			\footnotesize
			\begin{gothamtable}[gap=8pt]{lccc}
				\toprule
				& instructor & during-course & after-course \\
				\midrule
				\SMCmd{conditionalpause} & pause & -- & -- \\
				\SMCmd{deckitem}\SMCode{<n->} & from step $n$ & at once & at once \\
				\SMCmd{deckreveal}\SMCode{[n]} & from step $n$ & omitted & at once \\
				\SMCmd{deckrevealfinal}\SMCode{[n]} & from step $n$ & at once & at once \\
				\SMCmd{deckonly}\SMCode{<n>[v]} & step $n$ & if listed & if listed \\
				\bottomrule
			\end{gothamtable}
			\medskip

			\SMCmd{conditionalpause}\SMCode{[versions]} pauses in the listed versions (default: instructor); \SMCmd{deckuncover} and \SMCmd{deckvisible} work like \SMCmd{deckonly}; anything else: \SMCmd{ifSMInstructorVersion}, \SMCmd{ifSMDuringCourseVersion}, \SMCmd{ifSMAfterCourseVersion}
		\end{column}
	\end{columns}
\end{standardframe}
