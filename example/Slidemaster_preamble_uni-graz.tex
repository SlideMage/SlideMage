% Bibliography (SlideMage v5.32.0, 2026-10-07, direct instruction): the house
% citation format lives here, in the template, instead of being copied into
% every deck. A deck just calls \SMUseBibliography after \input{} of this file;
% it can redefine \SMBiblatexOptions / \SMBibliographyStyle before that call for a
% format of its own. (## below is # inside a macro body.)
% uniquename=init below silences a real (harmless) warning: biblatex's
% default uniquename=full tries to disambiguate same-surname authors by
% adding as much of their given name as needed, which conflicts with
% giveninits=true (which forces initials-only everywhere) -- biblatex
% auto-downgrades to uniquename=init on its own either way, this just does
% that up front instead of warning about it on every compile. Not needed
% for correctness (our own custom drivers below don't use the
% uniqueness-disambiguation machinery this controls at all), only for quiet.
\renewcommand{\SMBiblatexOptions}{style=authoryear, backend=biber, giveninits=true, uniquename=init, maxcitenames=3, mincitenames=1, maxbibnames=3, minbibnames=1}
\renewcommand{\SMBibliographyStyle}{%
% House citation format (not a stock biblatex style), built from scratch via
% custom bibliography drivers rather than tweaking authoryear's: every author
% as "Family, Initials." (not just the first), comma-separated with no "and"
% before the last one, "et al." only past three authors (maxcitenames=3/
% maxbibnames=3 above -- \fullcite goes through the bibliography-list code
% path, not the citation one, hence setting both), year unparenthesized,
% title in quotes (not italic), then a type-specific tail: edition+publisher
% for books, journal/volume(number)/pages/DOI for articles. Example targets:
%   Berk, J., DeMarzo, P., 2019. "Corporate Finance", 5th global, Pearson.
%   Fama, E., French, K. R., 1970. "Efficient Market Theory", Journal of
%   Finance 57(3), 115-120, https://doi.org/2929191992393.
\DeclareNameAlias{author}{family-given}
\renewcommand*{\multinamedelim}{\addcomma\space}
\renewcommand*{\finalnamedelim}{\addcomma\space}
% In-parenthesis citation mode (2026-07-27, direct instruction on IA2's
% slide he9: "(Clayson, 2005)" rather than \textcite's own "Clayson
% (2005)"); added here too, since this deck's bibliography block is kept
% in sync with decks/t-Uni-IA2-2026w.tex's and decks/#workdeck.tex's.
% \parencite already supplies the parentheses; biblatex's authoryear
% style separates name from year with a plain space, so this turns that
% separator into a comma. Deliberately the *default*-context declaration
% only: biblatex.def declares its own [textcite] and [bib,biblist]
% context variants of this same delimiter, so no \textcite call and no
% \fullcite/bibliography output anywhere changes. Verified directly
% rather than assumed from reading the style files (throwaway test deck,
% \textcite and \parencite rendered side by side with and without this
% line: \textcite output identical either way, only \parencite gained
% the comma).
\DeclareDelimFormat{nameyeardelim}{\addcomma\space}
% Plain (non-italic) title text: biblatex's inherited "standard" style
% italicizes the title field by default (\mkbibemph), which we don't want
% here since the title is already set off by quotes below. Overridden again
% specifically for @article, since the entrytype-specific format takes
% priority over the generic one above and would otherwise still auto-wrap
% article titles in its own curly quotes, doubling up with the
% \textquotedbl below (rock1986 in Slidemaster.bib is the first @article
% entry exercising this branch).
\DeclareFieldFormat{title}{##1}
\DeclareFieldFormat[article]{title}{##1}
% Drop the automatic "pp." prefix biblatex adds to page ranges by default --
% needed by rock1986 (Slidemaster.bib), the first @article entry.
\DeclareFieldFormat{pages}{##1}
% biblatex's own built-in isbn field format prepends a lowercase "isbn:"
% label of its own (confirmed directly on svv/IA2: rendered as
% "ISBN isbn: 978-...", doubling up with the book driver's own literal
% "ISBN" text below) -- stripped to the raw value only, same pattern as
% pages above.
\DeclareFieldFormat{isbn}{##1}
% Punctuation below is all *immediate* (\addcomma, \addperiod, ...), not
% biblatex's deferred \setunit{...} mechanism: \setunit only flushes its
% pending punctuation from inside the next \print... macro call, so a bare
% \textquotedbl (needed for a plain straight quote around the title, not
% \mkbibquote's language-sensitive curly one) sitting right after a \setunit
% prints *before* that deferred punctuation flushes -- e.g. "2023" then
% the quote, and only then the still-pending period, giving `2023". not
% `2023. -- immediate commands sidestep that entirely since every field used
% here is either always present (author/year/title/publisher) or already
% explicitly guarded by \iffieldundef (edition, doi/url, number).
\DeclareBibliographyDriver{book}{%
	\printnames{author}\addcomma\space
	\printfield{year}\addperiod\space
	\textquotedbl\printfield{title}\textquotedbl\addcomma\space
	\iffieldundef{edition}{}{\printfield{edition}\addcomma\space}%
	\printlist{publisher}%
	\iflistundef{location}{}{\addcomma\space\printlist{location}}%
	\iffieldundef{isbn}{}{\addcomma\space ISBN\space\printfield{isbn}}%
	\addperiod
	\newunit
	\finentry}
\DeclareBibliographyDriver{article}{%
	\printnames{author}\addcomma\space
	\printfield{year}\addperiod\space
	\textquotedbl\printfield{title}\textquotedbl\addcomma\space
	\printfield{journaltitle}\space
	\printfield{volume}%
	\iffieldundef{number}{}{(\printfield{number})}%
	\iffieldundef{pages}{}{\addcomma\space\printfield{pages}}%
	\iffieldundef{doi}%
		{\iffieldundef{url}{}{\addcomma\space\href{\thefield{url}}{\thefield{url}}}}%
		{\addcomma\space\href{https://doi.org/\thefield{doi}}{https://doi.org/\thefield{doi}}}%
	\addperiod
	\newunit
	\finentry}
% For an online/web resource (a Flickr photo, a web page, ...) that isn't a
% journal article or book -- CONVERSION_NOTES.md's "Plain-text citations"
% workflow rule is scoped to article/book specifically because neither of
% those two drivers fits a working paper (see that rule's own "not a
% working paper" carve-out); an online source is the same kind of gap, so
% it gets its own driver rather than being force-fit into @article (no
% journal/volume/pages) or left as a plain-text citation. Field usage:
% `organization` for the hosting site/platform (e.g. "Flickr"), `note` for
% anything after the URL that doesn't have its own dedicated field --
% access date and license both, as one string, since biblatex's own date
% type needs YYYY-MM-DD input and every other date-like field in this
% deck's own two other drivers above is already a plain string, not a
% real biblatex date, for the same reason. Example target:
%   Yamashita, J., 2016. "The Big Short", Flickr,
%   https://www.flickr.com/photos/bon_appetit/24901189243,
%   accessed: 07.07.2026, CC BY SA 2.0.
\DeclareBibliographyDriver{online}{%
	\printnames{author}\addcomma\space
	\printfield{year}\addperiod\space
	\textquotedbl\printfield{title}\textquotedbl\addcomma\space
	\iffieldundef{organization}{}{\printfield{organization}\addcomma\space}%
	\iffieldundef{url}{}{\href{\thefield{url}}{\thefield{url}}}%
	\iffieldundef{note}{}{\addcomma\space\printfield{note}}%
	\addperiod
	\newunit
	\finentry}
% For a working paper/technical report -- the gap the "online" driver's own
% comment above already flags ("neither of those two drivers fits a working
% paper"); filled in rather than left as plain text once one actually turned
% up (slides/y7q-Metallgesellschaft-Literature-current.tex's own Shetty/
% Manley citation, 2026-07-13). `institution`/`type` (e.g. "Iona College"/
% "working paper") print plain, unitalicized, after the title -- distinct
% from the "book"/"article" drivers' own dedicated field sets, since a
% working paper has neither a publisher nor a journal/volume/pages. Example
% target: Shetty, A., Manley, J., 2005. "Metallgesellschaft's Hedging
% Debacle." Iona College working paper.
%
% `\iflistundef`, not `\iffieldundef` (2026-07-23, found while adding new
% @report entries for slides/kuz-Shareholder-Pressure-current.tex/
% slides/aya-Behavioral-Effects-Sustainable-Investing-current.tex):
% `institution` is a biblatex *list* field (confirmed directly -- the
% resolved .bbl stores it as `\list{institution}{1}{...}`, not
% `\field{institution}{...}`), and `\iffieldundef` never recognizes a list
% field as defined regardless of its actual content, so this branch always
% took the empty case -- silently dropping the institution from *every*
% @report citation in this deck, including the very `shetty2005`/"Iona
% College" example this comment names as the target, which had never
% actually been rendering that way. Confirmed directly with an isolated
% minimal biblatex document (no beamer/gotham involved) reproducing the
% drop with `\iffieldundef` and fixing it with `\iflistundef`, both against
% the exact same two real .bib entries. `\printfield{institution}` also
% switched to `\printlist{institution}` to match (a plain `\printfield` on
% a list field is undefined behavior even when the `\if...undef` guard
% around it is fixed).
\DeclareBibliographyDriver{report}{%
	\printnames{author}\addcomma\space
	\printfield{year}\addperiod\space
	\textquotedbl\printfield{title}\textquotedbl\addcomma\space
	\iflistundef{institution}{}{\printlist{institution}\space}%
	\iffieldundef{type}{}{\printfield{type}}%
	\iffieldundef{doi}{}{%
		\iffieldundef{type}{\iflistundef{institution}{}{\unskip\addcomma\space}}{\addcomma\space}%
		\href{https://doi.org/\thefield{doi}}{https://doi.org/\thefield{doi}}}%
	\addperiod
	\newunit
	\finentry}
