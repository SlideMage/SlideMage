% SlideMage manual, chapter "Reveal steps": beamer's own overlay commands. The
% left column reveals live (four steps, in every deck version -- plain
% overlays do not know about deck versions; see the next slides). The
% \only line comes last, so that nothing below it moves when it appears.
\begin{standardframe}{Plain beamer reveals}
	{\small A slide can appear in steps, one PDF page per step. Beamer's own
	commands do this; the list below reveals itself as you page on.\par}
	\smallskip
	\begin{columns}[t,onlytextwidth]
		\begin{column}{0.46\linewidth}
			\begin{itemize}
				\item Always visible
				\item<2-> From step 2 on
				\item<3-> From step 3 on, \textcolor<4>{c_custom_red}{red} on step 4
			\end{itemize}
			\SMCodeFile{snippets/1dk-1.tex}
			\smallskip
			\only<4>{{\small Step 4 only, no space kept before\par}}
		\end{column}
		\begin{column}{0.5\linewidth}
			{\small Also \SMCmd{uncover}, \SMCmd{visible}, \SMCmd{alt}, \SMCode{\textbackslash only<2->\{...\}}\par}
			\medskip
			\begin{alertblock}{Two traps}
				\shrinklists{1}
				\begin{itemize}
					\item A reveal inside \SMCode{columns} or a \SMCode{minipage} takes its full height from step 1 on, and can push a figure beside it off the slide
					\item A picture placed on the page \emph{after} a \SMCmd{pause} stays hidden until the last step: put \SMCmd{onslide}\SMCode{<1->}\SMCmd{relax} before it
				\end{itemize}
			\end{alertblock}
		\end{column}
	\end{columns}
\end{standardframe}
