\setSlideVersion{1-0-1}
\begin{standardframe}{Options and other formats}
	\small
	\emph{Build options\ldots} in the arrow of \emph{Compile PDF} changes how PDFs are made; other outputs have entries of their own.
	\par\smallskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.63\linewidth}
			\textbf{Build options}
			\shrinklists{3}
			\begin{itemize}
				\item \emph{Include footer timestamp}: the compile date and time in the footer (off by default)
				\item \emph{Generate appendix TOC}: an agenda and clickable slide previews after every whole-deck PDF (also a deck setting)
				\item \emph{Pages replace theirs in the full PDF}: see \emph{Compiling single pages}
				\item \emph{Auto-open generated PDFs}: opens each PDF when it is finished
			\end{itemize}
			\textbf{Other outputs}
			\shrinklists{3}
			\begin{itemize}
				\item \emph{Export PPTX\ldots} (three-dot menu, \emph{This deck}): a PowerPoint file for recording a lecture, reveal steps fading in one by one
				\item \emph{Write transcript} (arrow of \emph{Compile PDF}): the accessible text version of a deck
			\end{itemize}
		\end{column}
		\begin{column}{0.34\linewidth}
			\centering
			{\setlength{\fboxsep}{0pt}\fcolorbox{c_primary_gray}{white}{\includegraphics[width=\dimexpr\linewidth-0.8pt\relax]{media/screenshot-slide-appendix-grid.png}}}\par
			{\scriptsize the appendix: a grid of slide previews\par}
		\end{column}
	\end{columns}
\end{standardframe}
