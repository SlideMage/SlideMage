% SlideMage manual, chapter "Accessibility": describing TikZ diagrams. The
% example is the diagram on slide 1ym, a real \tikzillustration.
\setSlideVersion{1-0-0}
\begin{standardframe}{Describing diagrams}
	\small
	A diagram drawn in LaTeX (TikZ) has no image file, so there is nothing to
	attach a description to. Wrap the drawing in a \SMCode{tikzillustration}
	with an ID of its own, and it becomes an image like any other: it appears in
	the Images tab, gets a description, and the AI model sees a rendering of it.
	\par\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.56\linewidth}
			\SMCodeFile{snippets/1pn-1.tex}
		\end{column}
		\begin{column}{0.41\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item The ID is the slide's ID plus \SMCode{-tikz}; the Images tab lists it as \SMCode{tikz:1ym-tikz}
				\item Example: the diagram on \SMLink{1ym}{\emph{Framework, project and app}} \mbox{(slide ID \emph{1ym})}
				\item Not for a picture placed on the page as an overlay, one inside another picture, or a mere frame around an image -- that image has its own description
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
