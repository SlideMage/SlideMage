% SlideMage setup layer and hook catalog (SlideMage v9.0.0, theme v9.7.0).
%
% A template \input{}s this file first, sets whatever it wants to change from
% the catalog below, then \input{}s SlideMage_base:
%
%   % SlideMage template name: My University
%   \input{SlideMage_setup}
%   \definecolor{c_primary_highlight}{HTML}{...}
%   \renewcommand{\SMInstituteLogoPath}{schemes/my_logo.pdf}
%   \setlength{\SMLogoHeight}{3ex}
%   \input{SlideMage_base}
%   ... additions only: dark-mode beamer colors, citation format, project macros
%
% Rules for a template:
%   1. Before this file: only \SMLayoutProfile (below; fira unless set) and \SMDarkMode
%      (written into the deck by SlideMage). Both decide what this file computes.
%   2. Between this file and SlideMage_base: \setlength, \renewcommand,
%      \definecolor and \colorlet on the names in this catalog. Every name
%      exists already, so a misspelt \renewcommand is an error instead of a
%      setting that silently does nothing.
%   3. After SlideMage_base: additions only.
%
% Decks never load the beamer theme themselves; this file does. A length given
% in ex or em is evaluated where the template sets it, in the normal body font,
% after the layout profile has loaded the fonts.
%
% Layout profiles
% ---------------
% Everything that depends on the font lives in a layout profile,
% SlideMage_layout_<name>.tex: the font files and how their weights map onto
% bold and italic, the code fonts, the math handling that goes with them, and
% the metrics the framework measures against (the ascender, the line spacing).
% A template picks one before this file, and switching it switches all of them
% together:
%
%   \newcommand{\SMLayoutProfile}{myfont}      % before \input{SlideMage_setup}; fira is the default
%
% SlideMage ships two profiles:
%   fira    SlideMage_layout_fira.tex: Fira Sans Light and Fira Mono, today's look and
%           the default (the fonts come with SlideMage's installer, not with TeX).
%   lmsans  SlideMage_layout_lmsans.tex: Latin Modern Sans and Latin Modern Mono, which
%           every TeX distribution has. A template uses it with
%             \newcommand{\SMLayoutProfile}{lmsans}   % before \input{SlideMage_setup}
% A project may keep its own profile next to its templates: the TeX search path finds
% SlideMage_layout_<name>.tex there. A template does not change a profile's numbers; it
% picks a profile or ships its own.
%
% What a profile defines (SlideMage_layout_fira.tex is the model; a test checks that every
% other profile defines each of these names):
%   fonts           the text font (\sfdefault with its weight mapping and \ttdefault), the
%                   mono font (\ttfamily), \SMCodeFont, \SMInlineCodeFont, \SMCardHeadingFont,
%                   and the math handling that goes with them
%   metrics         \SMFontAscender (the ascender in em, used where the font gives none of
%                   its own), \SMLinespread, \SMParskip
%   calibrations    lengths measured against the rendered PDF for this font, until the top
%                   placement of step 10 derives them as well:
%                     \SMContentTopShift \SMColumnsTopCorrection \SMColumnsTTopCompensation
% Nothing else: since v9.8.0 every other length is read from the font where it is used or is
% a hook of the catalog below (see 'Derived, not set').
%
% Recipe for a new profile:
%   1. Copy SlideMage_layout_fira.tex to SlideMage_layout_<name>.tex and change the
%      fonts (and \SMFontAscender, \SMLinespread, \SMParskip).
%   2. python tools/calibrate_font_profile.py --profile <name>
%      It compiles the calibration deck of tests/fixtures/template_stress/ with the
%      profile under the SlideMage template (the catalog's defaults) and reports, for each
%      length of the three above, the current value and the one the PDF suggests, and for
%      everything the base derives from the font, what the PDF shows against what the hooks
%      ask for.
%   3. Put the three suggestions into the profile's "Calibrations" section and run the tool
%      again until every line reads ok.
%   4. Compile the calibration deck with a template that picks the profile and run
%      tools/check_layout.py on the PDF (the stress-test template of that project does
%      this for lmsans). It reads the margins from the PDF, so a template with other
%      margins is checked against its own. Adjust until the report is clean.
% The last block of the tool's output reports what follows from the font and the template
% without a number of anyone's: where the title bar ends, the Top Margin, the footer number's
% and the footnotes' right edges, the footer number's ink in its box, the footnotes' gap to
% the footer and the section bar's height. A calibration that reads other catalog hooks (the
% pull-up of a frame the automatic Top Margin does not manage) is fitted to the catalog's
% defaults for those hooks.
%
% Derived, not set (SlideMage v9.7.0, v9.8.0)
% -------------------------------------------
% These lengths follow from the hooks and the font, and SlideMage_base computes them after the
% template (the first four) or where the font they depend on is in force (the rest). Slides may
% read them. A template or a profile that sets one gets an error, below, instead of a number
% that the next change to a hook or a font would silently contradict.
%   v9.7.0   \SMContentTopAlign (the Top Margin as a coordinate for overlays),
%            \SMTitleBarBottomAlign (where the title bar ends), \SMFootnoteRightShift (a
%            footnote's right edge against the text margin), \SMFooterRightShift (the same for
%            the frame number)
%   v9.8.0   \SMFooterTextDrop (the frame number's ink centred in the footer box),
%            \SMFootlineVOffset (the footnotes' descender \SMFootnoteFooterGap above the
%            footer box), \SMLabelItemWidth, \SMLabelItemSep (the widest label of the lists
%            with "a)" / "1." labels and the gap to the text), \SMEnumerateNestedMinNumber,
%            \SMEnumerateNestedLabelSep, \SMEnumerateNestedLeftMargin, \SMEnumerateLabelBox
%            (the numeral column of a nested enumerate), and the section frame's four
%            corrections \SMSectionframeRefDepth, \SMSectionframeBottomGap,
%            \SMSectionframeBottomPadCorrection, \SMSectionframeHeightCorrection (the frame
%            places its title by the title's own box, with \SMSectionframeTitleGap)

% The derived names, as macros that stop with an error when a template or a profile sets
% or redefines them. SlideMage_base compares them with the copies kept here, replaces them with
% the lengths it derives, and reports a name that was redefined.
\makeatletter
\newcommand{\SM@derivedname}[1]{\PackageError{SlideMage}{#1 is derived, not set}{%
	SlideMage_base computes #1 from the hooks and the font (see 'Derived, not set' in
	SlideMage_setup.tex). Remove the line that sets it from the template or the layout profile;
	to move what it places, change the hook it follows.}}

	\newcommand{#1}{\SM@derivedname{\string#1}}%
	\expandafter\let\csname SM@guard@\expandafter\@gobble\string#1\endcsname#1}

% Top Margin
% ---------------------------------------------------------------------------
% The distance from the top of the page to the top of a frame's first line (its
% line box: the baseline minus the font's ascender), in bp, the unit of the PDF.
% SlideMage measures every frame at shipout and moves its body until the first line
% sits there. A dimension with a unit; "bp" is read exactly. SlideMage_base.tex
% hands it, with the layout profile's ascender, to SlideMage_measure.lua.
%
% Everything that sits on the Top Margin follows this one setting: the overlay coordinate
% \SMContentTopAlign (corner figures, \SMFit, floating blocks, frames with a side image) is
% derived from it by the base, as are the bottom edge of the title bar
% (\SMTitleBarBottomAlign, from the paddings, the frame title font and the logo height) and
% the right edges of footer and footnotes (from the text margin and the logo's right padding).
% A template sets none of those.

% ---------------------------------------------------------------------------
% Optional theme features (all off: nothing visible changes until you use one)
% ---------------------------------------------------------------------------
% The SlideMage theme (beamerthemeSlideMage.sty) keeps four features of the
% Gotham theme it was forked from. Use their commands here in the template,
% between \input{SlideMage_setup} and \input{SlideMage_base}, never in a group.
% They read the \part / \section / \subsection structure of the deck.
%
% Mini frames: one small mark per frame in a bar.
%   \SMMiniFramesPosition{head|foot|left|right|none}     where the bar goes (none)
%   \SMMiniFramesSpreading{spreading|centering|left|right}  how the marks are spread
%   \SMMiniFramesSections{none|section|secsubsection}    names next to the marks
%                                                         (section: beamer's own)
%   \SMMiniFramesCompress{true|false}                    one row per section (true)
%   \setbeamertemplate{mini frames}[SMBox|SMMiniBox|SMMiniBullet]   the mark (SMMiniBullet)
%   \setbeamertemplate{mini frames bundle}[SMMini|SMNano]           the marks' set (SMMini)
%
% Progress bar: a bar that fills with the frame number.
%   \SMProgressBarPosition{head|frametitle|foot|left|right|none}   where it goes (none)
%   \SMProgressBarStyle{rectangle|rounded box}            its look (rectangle)
%   \SMProgressBarDirection{tlbr|brlt}                    the direction it fills (tlbr)
%   \SMProgressBarMiniFrames{true|false}                  mini frames inside the bar (false)
%   \setlength{\SMProgressHeadFootLineheight}{1.2ex}      its thickness
%   Left and right need a sidebar width: \setbeamersize{sidebar width left=1cm}.
%   The bar counts to 100 frames (\inserttotalframenumber is 100, as in Gotham).
%
% Divider frames: beamer inserts a frame at each \part, \section, \subsection
% or \subsubsection (SlideMage's own \sectionframe is a different thing).
%   \SMDividerFrames{part,section,subsection,subsubsection|none}   the levels (none)
%   \SMDividerLook{SMSimple|SMProgressbar|SMSplitvertSimple|SMSplitvertProgressbar|SMProgressvert}
%                                                         how they look (SMProgressbar)
%   \renewcommand{\SMSectionPageOptions}{...}              frame options per level (noframenumbering);
%                                                         likewise \SMPartPageOptions,
%                                                         \SMSubsectionPageOptions, ...
%   \setlength{\SMSectionHOffset}{0pt}                     shifts the text of a divider
%
% Watermark: a text behind the frames.
%   \SMWatermarkDefault{true|false}   on every frame (false); [watermark] and
%                                     [nowatermark] switch single frames
%   \SMWatermarkTemplate{SMDraft}     the background template to use; define your own
%                                     with \defbeamertemplate{background}{<name>}{...}
