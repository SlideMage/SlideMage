\setSlideVersion{1-1-1+}
\begin{standardframe}{Preview, source and split}
	\small
	The middle column shows the focused slide as \emph{Preview}, \emph{Split}, \emph{Source} or \emph{Details}, chosen above it. In \emph{Source} you write the LaTeX directly.
	\par\smallskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.52\linewidth}
			{\setlength{\fboxsep}{0pt}\fcolorbox{c_primary_gray}{white}{\includegraphics[width=\dimexpr\linewidth-0.8pt\relax]{media/screenshot-slidemage-slide-editor.png}}}
		\end{column}
		\begin{column}{0.45\linewidth}
			\shrinklists{2}
			\begin{itemize}
				\item \textbf{Writing.} Buttons above the source insert styles, footnotes and colors; \emph{Insert} adds lists, blocks, commands, links and code files; \emph{Cite} searches the bibliography. Edits are saved as you type; \emph{Update version} asks how big the change is and makes a version.
				\item \textbf{Checking.} A line below the toolbar names mistakes as you type; \emph{Ask AI to fix} hands them to the chat.
				\item \textbf{Seeing the result.} \emph{Render} redraws the Preview; its arrow renders the editor's text, saved or not (\SMCode{Ctrl+R}). \emph{Split} shows both side by side.
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
