% SlideMage manual, chapter "Plugins": first start and the Plugins sheet.
\setSlideVersion{1-0-0}
\begin{standardframe}{Installing plugins}
	\small
	Open \emph{File} $\rightarrow$ \emph{Settings\ldots} $\rightarrow$
	\emph{Plugins}, or \emph{Help} $\rightarrow$ \emph{Plugins\ldots}
	\par\medskip
	\shrinklists{1}
	\begin{itemize}
		\item \textbf{First start}: a sheet offers plugins once per computer, if the list has any. Nothing is ticked; recommendations come first. \emph{Not now} closes it
		\item \textbf{Available}: read the summary, needs and network contacts, then \emph{Install}. \emph{Check for plugins} refreshes the list
		\item \textbf{Install from file\ldots}: choose a \SMCode{.smplugin} file and confirm its warning sheet. A matching \SMCode{name.smplugin.sha256} file must be beside it
		\item \textbf{Installed}: version, state, \emph{Enabled}, \emph{Update to\ldots} and \emph{Remove}; \emph{Details\ldots} explains a load failure
		\item \textbf{Restart SlideMage} after changes: plugins load once, before the window opens
	\end{itemize}
\end{standardframe}
