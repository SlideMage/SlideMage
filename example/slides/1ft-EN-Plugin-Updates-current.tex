% SlideMage manual, chapter "Plugins": the app's update mode also controls plugins.
\setSlideVersion{1-0-0}
\begin{standardframe}{Plugin updates}
	\small
	The start-up update check includes installed plugins from the SlideMage
	list, at most once a day. It follows \emph{File} $\rightarrow$
	\emph{Settings\ldots} $\rightarrow$ \emph{General} $\rightarrow$ \emph{Updates}.
	\par\medskip
	\shrinklists{1}
	\begin{itemize}
		\item \textbf{Don't check}: no plugin update check
		\item \textbf{Ask} (the default): names plugins with a newer release; choose \emph{Update to\ldots} in Plugins to install one
		\item \textbf{Install automatically}: installs newer releases automatically
		\item \textbf{Restart} for an update to take effect. A plugin run from a checkout is not updated
	\end{itemize}
\end{standardframe}
