% SlideMage manual, chapter "Installing SlideMage": Ollama and the models
% SlideMage's AI features use. Everything else in SlideMage works without it.
\setSlideVersion{1-0-1+}
\begin{standardframe}{Local AI models: Ollama}
	\small
	With the \emph{Ollama} plugin (\emph{Settings} $\rightarrow$ \emph{Extensions}), descriptions,
	translations, syntax fixes and the AI chat can use models
	that run on your own computer -- nothing is sent
	online. Without the plugin or Ollama, everything else works as usual.
	\par
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.49\linewidth}
			\shrinklists{1}
			\textbf{Installation}
			\begin{itemize}
				\item The Windows installer from \SMCode{ollama.com/download}; Ollama then runs in the background
				\item Download models in a terminal, e.g., \SMCode{ollama pull gemma4:12b}
				\item In SlideMage: \emph{Settings} $\rightarrow$ \emph{AI}, section \emph{Ollama}, lists the installed models; \emph{AI setup} chooses one per task, kept on this computer for every project
			\end{itemize}
		\end{column}
		\begin{column}{0.47\linewidth}
			\shrinklists{1}
			\textbf{Which models}
			\begin{itemize}
				\item \emph{Image descriptions}: a vision model, e.g., \SMCode{gemma4:12b} or \SMCode{qwen3-vl:32b}
				\item \emph{Translation}: e.g., \SMCode{translategemma:12b}
				\item \emph{Syntax check}: e.g., \SMCode{qwen3-coder}
				\item \emph{Chat assistant}: a model that can call tools, e.g., \SMCode{qwen3}; only such models are listed
				\item Larger models write better but need a graphics card with enough memory; on a processor alone they are slow
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
