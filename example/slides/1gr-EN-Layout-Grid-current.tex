	\begin{minipage}[t]{0.46\linewidth}\vspace{0pt}
		{\small\SMLineTop You rarely need \SMCmd{vspace}: the framework puts every
		frame's first line on the Top Margin and keeps content between the
		margins. Place your own drawings on these lengths, not on measured
		numbers.\par}
		\medskip
		\shrinklists{1}
		\begin{itemize}
			\item \SMCmd{SMManualTopMargin} before a frame keeps the default position; \SMCmd{SMAutoTopMarginfalse} places every frame of a deck that way
			\item More lengths: \SMCmd{SMSideImageFrameWidth}, \SMCmd{SMImageBorderWidth}, \SMCmd{SMItemStep}
			\item \SMCode{tools/check\_layout.py} measures a compiled PDF
		\end{itemize}
	\end{minipage}
