% SlideMage manual, appendix: installation and the tools not covered in the
% chapters (README: "Quick start", "Releases and updates"; PROJECT_OVERVIEW).
\begin{standardframe}{Setup and more tools}
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.48\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item \textbf{Needed}: MiKTeX or TeX Live with LuaLaTeX, \SMCode{biber} and \SMCode{latexmk} (on Windows its Perl comes with Git for Windows; SlideMage finds it)
				\item \textbf{SlideMage}: the released program, or \SMCode{pip install -e .} from the repository
				\item \textbf{Optional}: Ollama with a vision model, for image descriptions and the AI helpers
				\item \textbf{Which project}: \SMCode{--project}, \SMCode{\$SLIDEMASTER\_PROJECT}, the folder with \SMCode{slide\_ids.csv} above the current one, or the last one opened
			\end{itemize}
		\end{column}
		\begin{column}{0.48\linewidth}
			\shrinklists{1}
			\begin{itemize}
				\item \textbf{Import PPTX/ODP} turns a presentation's slides into new slides of a deck
				\item \textbf{Archive unused content} (Housekeeping)
				\item \textbf{Updates}: Preferences $\rightarrow$ Updates (don't check / ask / install); \SMCode{.slidemage/project.json} records the format of a project
				\item \textbf{Preferences}: dark mode, the default template for single-slide renders, AI models and prompts
				\item \textbf{Older projects}: \SMCode{python -m slidemage.sm\_rename <project>} renames \SMCode{\textbackslash gotham...} commands to \SMCode{\textbackslash SM...}
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
