% SlideMage manual, chapter "Reveal steps": beamer's own overlay commands. The
% left column reveals live (four steps, in every deck version -- plain
% overlays do not know about deck versions; see the next slides). The
% \only line comes last, so that nothing below it moves when it appears.
% Split in two on 2026-10-07: the other commands and the two traps are on 1un.
\setSlideVersion{1-1-0}
\begin{standardframe}{Plain beamer reveals}
	{\small A slide can appear in steps, one PDF page per step. Beamer's own
	commands do this, and they work in every deck version: the list below reveals
	itself as you page on.\par}
	\smallskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.46\linewidth}
			\begin{itemize}
				\item Always visible
				\item<2-> From step 2 on
				\item<3-> From step 3 on, \textcolor<4>{c_custom_red}{red} on step 4
			\end{itemize}
			\SMCodeFile{snippets/1dk-1.tex}
			\smallskip
			\only<4>{{\small Step 4 only, no space kept before\par}}
		\end{column}
		\begin{column}{0.5\linewidth}
			\shrinklists{2}
			\begin{itemize}
				\item \SMCode{\textbackslash item<2->} shows an item from step 2 on; it stays visible afterwards
				\item A command with a step after it applies only there: \SMCode{\textbackslash textcolor<4>\{red\}\{...\}} is red on step 4 only
				\item \SMCode{\textbackslash only<4>\{...\}} shows its text on step 4 only and keeps no space for it on the other steps, so put it last
				\item \SMCmd{pause} makes everything after it appear on the next step
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
