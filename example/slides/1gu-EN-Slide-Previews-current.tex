\setSlideVersion{1-0-0+}
\begin{standardframe}{Slide previews}
	\small
	SlideMage keeps a picture of every slide for the slide list and the \emph{Preview}, rendered in the background while you are not using it, so decks open with their previews ready.
	\par\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.37\linewidth}
			{\setlength{\fboxsep}{0pt}\fcolorbox{c_primary_gray}{white}{\includegraphics[width=\dimexpr\linewidth-0.8pt\relax]{media/screenshot-slidemage-slide-preview.png}}}
		\end{column}
		\begin{column}{0.60\linewidth}
			\shrinklists{2}
			\begin{itemize}
				\item \emph{Preview} shows the saved picture at once; chips say what it is not showing: a pinned version, another language, a missing translation, a view that skips the slide
				\item \emph{preview old} in a list row: the source is newer; \emph{Render} makes a fresh one
				\item \emph{Background previews}: the dial under the slide shows how much is ready, the chip what is left; \emph{Settings} $\rightarrow$ General sets speed and disk limit
				\item The mouse wheel steps through reveal steps, then slides; the bar under the slide shows each deck version's steps
				\item \emph{Housekeeping} $\rightarrow$ \emph{Update slide previews} renders all slides
			\end{itemize}
		\end{column}
	\end{columns}
\end{standardframe}
